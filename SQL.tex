CREATE PROCEDURE ExcluirUsuarioComPermissoes
    @UsuarioId INT
AS
BEGIN
    BEGIN TRANSACTION;
    
    BEGIN TRY
        -- Deletar as permissões associadas ao usuário
        DELETE FROM PermissaoMenu WHERE UsuarioId = @UsuarioId;
        
        -- Deletar o usuário
        DELETE FROM Usuarios WHERE Id = @UsuarioId;
        
        COMMIT TRANSACTION; -- Confirmar a transação
    END TRY
    BEGIN CATCH
        ROLLBACK TRANSACTION; -- Reverter a transação em caso de erro
        
        -- Tratar o erro e lançar uma exceção
        DECLARE @ErroNumero INT, @ErroMensagem NVARCHAR(4000), @ErroEstado INT;
        
        SELECT 
            @ErroNumero = ERROR_NUMBER(),
            @ErroMensagem = ERROR_MESSAGE(),
            @ErroEstado = ERROR_STATE();
        
        RAISERROR (@ErroMensagem, 16, 1); -- Lançar uma exceção com o número e a mensagem do erro
        
        -- Opcionalmente, você pode registrar o erro em uma tabela de logs ou executar outras ações
    END CATCH
END;
